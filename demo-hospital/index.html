[ApiController]
[Route]("api/[controller]")
public class StaffController : ControllerBase
{
    private readonly HospitalDbContext _db;
    public StaffController(HospitalDbContext db) => _db = db;

    [HttpGet]
    public async Task<IActionResult> GetAllStaff([FromQuery] StaffRole? role = null)
    {
        var query = _db.StaffMembers.AsNoTracking().AsQueryable();
        if (role.HasValue) query = query.Where(s => s.Role == role.Value);
        return Ok(await query.ToListAsync());
    }

    [HttpPut("{id}")]
    public async Task<IActionResult> UpdateStaff(int id, [FromBody] CreateStaffDto dto)
    {
        var staff = await _db.StaffMembers.FindAsync(id);
        if (staff == null) return NotFound();

        staff.FullName = dto.FullName;
        staff.Role = dto.Role;
        staff.Department = dto.Department;
        staff.MonthlySalary = dto.MonthlySalary;
        staff.Shift = dto.Shift;

        await _db.SaveChangesAsync();
        return Ok(new { message = "Staff member updated successfully." });
    }

    [HttpDelete("{id}")]
    public async Task<IActionResult> DeleteStaff(int id)
    {
        var staff = await _db.StaffMembers.FindAsync(id);
        if (staff == null) return NotFound();

        _db.StaffMembers.Remove(staff);
        await _db.SaveChangesAsync();
        return Ok(new { message = $"Staff '{staff.FullName}' removed." });
    }
}

[HttpPost("disburse")]
public async Task<IActionResult> DisburseSalary([FromBody] DisburseSalaryDto dto)
{
    var record = await _db.SalaryRecords
        .Include(r => r.StaffMember)
        .FirstOrDefaultAsync(r => r.StaffMemberId == dto.StaffMemberId && r.MonthYear == dto.MonthYear);

    record.Status = PayrollStatus.Paid;
    record.PaidDate = DateTime.UtcNow;
    record.TransactionReference = $"TX-PAY-{DateTime.UtcNow:yyyyMMdd}-{Guid.NewGuid().ToString()[..6]}";

    await _db.SaveChangesAsync();
    return Ok(new { Status = "Paid", TxRef = record.TransactionReference });
}

public class HospitalDbContext : DbContext
{
    public HospitalDbContext(DbContextOptions<HospitalDbContext> opt) : base(opt) { }

    public DbSet<StaffMember> StaffMembers => Set<StaffMember>();
    public DbSet<SalaryRecord> SalaryRecords => Set<SalaryRecord>();
    public DbSet<Patient> Patients => Set<Patient>();
    public DbSet<Bed> Beds => Set<Bed>();
}
Location: C:\CarePulseERP\Database\schema.sql Relational Database 
CREATE TABLE StaffMembers (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    FullName VARCHAR(120) NOT NULL,
    Role INT NOT NULL COMMENT 'Doctor, Nurse, Sweeper, Receptionist, Pharmacist',
    Department VARCHAR(80) NOT NULL,
    MonthlySalary DECIMAL(18,2) NOT NULL,
    IsActive BOOLEAN DEFAULT TRUE
);

CREATE TABLE SalaryRecords (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    StaffMemberId INT NOT NULL,
    MonthYear VARCHAR(20) NOT NULL,
    BaseSalary DECIMAL(18,2) NOT NULL,
    Status INT DEFAULT 1,
    FOREIGN KEY (StaffMemberId) REFERENCES StaffMembers(Id)
);
